/**
 * CameraX 相机控制器：预览 + 帧分析双通道。
 * 帧分析流既驱动 MediaPipe 姿态推理，也为零快门延迟拍照提供最新帧。
 */
class CameraController(
    private val context: Context,
    private val lifecycleOwner: LifecycleOwner,
    private val previewView: PreviewView,
    private val onFrame: (androidx.camera.core.ImageProxy) -> Unit,
) {
    var lensFacing: Int = CameraSelector.LENS_FACING_BACK
        private set

    private var cameraProvider: ProcessCameraProvider? = null
    private val analysisExecutor: ExecutorService = Executors.newSingleThreadExecutor()

    fun start() {
        val future = ProcessCameraProvider.getInstance(context)
        future.addListener({
            cameraProvider = future.get()
            bind()
        }, ContextCompat.getMainExecutor(context))
    }

    fun switchCamera() {
        lensFacing = if (lensFacing == CameraSelector.LENS_FACING_BACK) {
            CameraSelector.LENS_FACING_FRONT
        } else {
            CameraSelector.LENS_FACING_BACK
        }
        bind()
    }

    fun isFront() = lensFacing == CameraSelector.LENS_FACING_FRONT

    private fun bind() {
        val provider = cameraProvider ?: return

        val preview = Preview.Builder().build().also {
            it.setSurfaceProvider(previewView.surfaceProvider)
        }

        val resolutionSelector = ResolutionSelector.Builder()
            .setResolutionStrategy(
                ResolutionStrategy(
                    android.util.Size(1280, 720),
                    ResolutionStrategy.FALLBACK_RULE_CLOSEST_HIGHER_THEN_LOWER
                )

----- 第 9 页 -----
            )
            .build()

        val analysis = ImageAnalysis.Builder()
            .setResolutionSelector(resolutionSelector)
            .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
            .setOutputImageFormat(ImageAnalysis.OUTPUT_IMAGE_FORMAT_YUV_420_888)
            .build()
        analysis.setAnalyzer(analysisExecutor) { image ->
            onFrame(image)
            image.close()
        }

        provider.unbindAll()
        provider.bindToLifecycle(
            lifecycleOwner,
            CameraSelector.Builder().requireLensFacing(lensFacing).build(),
            preview,
            analysis,
        )
    }

    fun shutdown() {
        cameraProvider?.unbindAll()
        analysisExecutor.shutdown()
    }
}

package com.poseadvisor.app.camera

import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.ImageFormat
import android.graphics.Matrix
import android.graphics.Rect
import android.graphics.YuvImage
import androidx.camera.core.ImageProxy
import java.io.ByteArrayOutputStream

/**
 * 零快门延迟拍照的核心：最新预览帧的 NV21 字节环形缓冲。
 * 帧到达时仅做一次内存拷贝（毫秒级），按下快门时无需再向相机请求新帧。
 */
class FrameBuffer {

    data class Frame(
        val nv21: ByteArray,
        val width: Int,

----- 第 10 页 -----
        val height: Int,
        val rotationDegrees: Int,
        val timestampMs: Long,
    )

    @Volatile
    var latest: Frame? = null
        private set

    /** 从 ImageProxy 拷贝 YUV_420_888 数据为 NV21 并发布为最新帧 */
    fun accept(image: ImageProxy) {
        val w = image.width
        val h = image.height
        val nv21 = ByteArray(w * h * 3 / 2)
        yuv420ToNv21(image, nv21)
        latest = Frame(nv21, w, h, image.imageInfo.rotationDegrees, System.currentTimeMillis())
    }

    /** 将最新帧解码为处理过旋转的 Bitmap（可选缩放） */
    fun decodeLatest(maxDim: Int = 0): Bitmap? {
        val f = latest ?: return null
        val yuv = YuvImage(f.nv21, ImageFormat.NV21, f.width, f.height, null)
        val out = ByteArrayOutputStream()
        yuv.compressToJpeg(Rect(0, 0, f.width, f.height), 92, out)
        var bmp = BitmapFactory.decodeByteArray(out.toByteArray(), 0, out.size()) ?: return null
        if (f.rotationDegrees != 0) {
            val m = Matrix().apply { postRotate(f.rotationDegrees.toFloat()) }
            bmp = Bitmap.createBitmap(bmp, 0, 0, bmp.width, bmp.height, m, true)
        }
        if (maxDim > 0 && maxOf(bmp.width, bmp.height) > maxDim) {
            val scale = maxDim.toFloat() / maxOf(bmp.width, bmp.height)
            val m = Matrix().apply { postScale(scale, scale) }
            bmp = Bitmap.createBitmap(bmp, 0, 0, bmp.width, bmp.height, m, true)
        }
        return bmp
    }

    companion object {
        /** 兼容任意 rowStride / pixelStride 的 YUV_420_888 → NV21 转换 */
        fun yuv420ToNv21(image: ImageProxy, out: ByteArray) {
            val w = image.width
            val h = image.height
            val yPlane = image.planes[0]
            val uPlane = image.planes[1]
            val vPlane = image.planes[2]

            // Y 平面
            var offset = 0
            val yBuffer = yPlane.buffer
            val yRowStride = yPlane.rowStride

----- 第 11 页 -----
            if (yRowStride == w) {
                yBuffer.get(out, 0, w * h)
                offset = w * h
            } else {
                val row = ByteArray(w)
                for (r in 0 until h) {
                    yBuffer.position(r * yRowStride)
                    yBuffer.get(row, 0, w)
                    System.arraycopy(row, 0, out, offset, w)
                    offset += w
                }
            }

            // VU 平面（NV21 为 V 在前）
            val chromaH = h / 2
            val chromaW = w / 2
            val vBuffer = vPlane.buffer
            val uBuffer = uPlane.buffer
            val vRowStride = vPlane.rowStride
            val uRowStride = uPlane.rowStride
            val vPixelStride = vPlane.pixelStride
            val uPixelStride = uPlane.pixelStride

            for (r in 0 until chromaH) {
                for (c in 0 until chromaW) {
                    out[offset++] = vBuffer.get(r * vRowStride + c * vPixelStride)
                    out[offset++] = uBuffer.get(r * uRowStride + c * uPixelStride)
                }
            }
        }
    }
}

class PoseLandmarkerHelper(
    context: Context,

    private var landmarker: PoseLandmarker? = null

    init {
        val baseBuilder = BaseOptions.builder()
            .setModelAssetPath("pose_landmarker_lite.task")
        try {
            baseBuilder.setDelegate(Delegate.GPU)
            landmarker = create(context, baseBuilder.build())
        } catch (e: Exception) {
            Log.w(TAG, "GPU 委托不可用，回退 CPU", e)
            try {
                baseBuilder.setDelegate(Delegate.CPU)
                landmarker = create(context, baseBuilder.build())
            } catch (e2: Exception) {
                Log.e(TAG, "PoseLandmarker 初始化失败", e2)
            }
        }
    }

    private fun create(context: Context, base: BaseOptions): PoseLandmarker {
        val options = PoseLandmarker.PoseLandmarkerOptions.builder()
            .setBaseOptions(base)
            .setRunningMode(RunningMode.LIVE_STREAM)
            .setNumPoses(1)
            .setMinPoseDetectionConfidence(0.5f)
            .setMinPosePresenceConfidence(0.5f)
            .setMinTrackingConfidence(0.5f)
            .setResultListener { result, _ ->
                PoseMapper.to75d(result)?.let(onResult)
            }
            .setErrorListener { e -> Log.e(TAG, "推理错误: ${e.message}") }
            .build()
        return PoseLandmarker.createFromOptions(context, options)
    }

/**
 * 姿势比对引擎：
 * 将用户实时姿态与推荐引导姿态做尺度/位置无关的归一化后，
 * 按肢体段方向角误差加权计算匹配度（0-100），并给出最大偏差部位的纠正提示。
 */
object PoseComparator {

    data class CompareResult(
        val score: Int,                       // 0-100 匹配度
        val segmentOk: BooleanArray,          // 每肢体段是否达标（用于着色）
        val hint: String?,                    // 最需要纠正的部位提示
    )

    private const val OK_THRESHOLD_RAD = 0.28f   // 单段达标阈值（约16°）
    private const val SCORE_K = 2.6f             // 误差→分数的衰减系数

    /** 归一化：骨盆平移到原点，按躯干长度（骨盆→颈）缩放，消除距离与身高差异 */
    fun normalize(pose: FloatArray): FloatArray {
        val out = pose.copyOf()
        val px = pose.jx(PoseModel.PELVIS)
        val py = pose.jy(PoseModel.PELVIS)
        val torso = sqrt(
            (pose.jx(PoseModel.NECK) - px) * (pose.jx(PoseModel.NECK) - px) +
                (pose.jy(PoseModel.NECK) - py) * (pose.jy(PoseModel.NECK) - py)
        ).coerceAtLeast(1e-4f)
        for (i in 0 until PoseModel.NUM_JOINTS) {
            out[i * 3] = (pose[i * 3] - px) / torso
            out[i * 3 + 1] = (pose[i * 3 + 1] - py) / torso
        }
        return out
    }

    /** 生成用于屏幕展示的引导骨架：骨盆置于 (0.5, 0.55)，躯干高度约 0.30 */
    fun guideForDisplay(guide: FloatArray): FloatArray {
        val n = normalize(guide)
        val scale = 0.30f
        for (i in 0 until PoseModel.NUM_JOINTS) {
            n[i * 3] = n[i * 3] * scale + 0.5f
            n[i * 3 + 1] = n[i * 3 + 1] * scale + 0.55f
        }
        return n
    }

    fun compare(userPose: FloatArray, guidePose: FloatArray): CompareResult {
        val u = normalize(userPose)

----- 第 17 页 -----
        val g = normalize(guidePose)

        val segs = PoseModel.COMPARE_SEGMENTS
        val segOk = BooleanArray(segs.size)
        var weightedSum = 0f
        var weightTotal = 0f
        var worstIdx = -1
        var worstErr = -1f

        for (i in segs.indices) {
            val s = segs[i]
            val angU = atan2(u.jy(s.b) - u.jy(s.a), u.jx(s.b) - u.jx(s.a))
            val angG = atan2(g.jy(s.b) - g.jy(s.a), g.jx(s.b) - g.jx(s.a))
            var err = abs(angU - angG)
            if (err > Math.PI.toFloat()) err = (2 * Math.PI).toFloat() - err
            segOk[i] = err < OK_THRESHOLD_RAD
            weightedSum += err * s.weight
            weightTotal += s.weight
            if (err > worstErr) { worstErr = err; worstIdx = i }
        }

        val mean = weightedSum / weightTotal
        val score = (100f * exp(-SCORE_K * mean)).toInt().coerceIn(0, 100)

        val hint = if (worstIdx >= 0 && worstErr > OK_THRESHOLD_RAD) {
            val s = segs[worstIdx]
            // 用远端关节的相对位移给出方向提示
            val dx = g.jx(s.b) - u.jx(s.b)
            val dy = g.jy(s.b) - u.jy(s.b)
            val dir = if (abs(dy) >= abs(dx)) {
                if (dy < 0) "上" else "下"
            } else {
                if (dx < 0) "左" else "右"
            }
            "请把${s.partName}向${dir}调整"
        } else null

        return CompareResult(score, segOk, hint)
    }
}

/**
 * 轻量场景识别：对最新预览帧做亚采样统计（亮度/绿色占比/天空占比），
 * 毫秒级完成，为姿势推荐选择默认类别。用户可随时手动切换类别。
 */
object SceneClassifier {

    data class SceneHint(val label: String, val suggestedCategory: String)

    fun classify(frame: FrameBuffer.Frame): SceneHint {
        val data = frame.nv21
        val w = frame.width
        val h = frame.height

        var brightSum = 0L
        var greenCount = 0
        var skyCount = 0
        var total = 0
        var topTotal = 0

        // 亚采样 Y 平面（每 8 行每 8 列取一个点）
        var r = 0
        while (r < h) {
            var c = 0
            while (c < w) {
                val y = data[r * w + c].toInt() and 0xFF
                brightSum += y
                // 用 UV 粗判绿色/蓝色（NV21：VU 交错）
                val uvIndex = w * h + (r / 2) * w + (c / 2) * 2
                if (uvIndex + 1 < data.size) {
                    val v = data[uvIndex].toInt() and 0xFF
                    val u = data[uvIndex + 1].toInt() and 0xFF
                    // 绿色：U、V 均偏低；天空蓝：U 高 V 中低
                    if (u < 100 && v < 110 && y in 40..220) greenCount++
                    if (r < h / 3) {
                        topTotal++
                        if (u > 130 && v < 130 && y > 80) skyCount++
                    }
                }
                total++

----- 第 20 页 -----
                c += 8
            }
            r += 8
        }

        if (total == 0) return SceneHint("日常街拍", "standing")
        val brightness = brightSum.toFloat() / total / 255f
        val greenRatio = greenCount.toFloat() / total
        val skyRatio = if (topTotal > 0) skyCount.toFloat() / topTotal else 0f

        return when {
            greenRatio > 0.25f -> SceneHint("户外自然", "standing")
            skyRatio > 0.30f -> SceneHint("开阔天空", "dynamic")
            brightness < 0.33f -> SceneHint("室内弱光", "sitting")
            brightness > 0.72f -> SceneHint("明亮通透", "gesture")
            else -> SceneHint("日常街拍", "standing")
        }
    }
}

def is_valid_pose(joints: np.ndarray) -> bool:
    """过滤明显异常的采样结果。"""
    if not np.isfinite(joints).all():
        return False
    xy = joints[:, :2]
    # 关键点应大体落在画面附近（允许少量越界）
    if (xy < -0.6).any() or (xy > 1.6).any():
        return False
    pelvis, neck = joints[0], joints[4]
    torso = np.linalg.norm(neck[:2] - pelvis[:2])
    if torso < 0.05:  # 人体过小/坍缩
        return False

def normalize_pose(joints: np.ndarray) -> np.ndarray:
    """骨盆对齐原点 + 躯干长度归一，消除画面位置与比例差异（与 App 端比对一致）。"""
    out = joints.copy()
    pelvis = out[0, :2].copy()
    torso = np.linalg.norm(out[4, :2] - pelvis) + 1e-6
    out[:, :2] = (out[:, :2] - pelvis) / torso
    return out


def kmeans_numpy(feats: np.ndarray, k: int, iters: int = 50, seed: int = 0):
    """轻量 KMeans（k-means++ 初始化），避免引入 sklearn 依赖。"""
    rng = np.random.default_rng(seed)
    n = len(feats)
    # k-means++ 初始化
    centers = [feats[rng.integers(n)]]
    for _ in range(k - 1):
        d = np.min(np.linalg.norm(feats[:, None, :] - np.array(centers)[None], axis=-1), axis=1)
        probs = d ** 2
        probs /= probs.sum() + 1e-12
        centers.append(feats[rng.choice(n, p=probs)])
    centers = np.array(centers)
    labels = np.zeros(n, dtype=int)
    for _ in range(iters):
        dist = np.linalg.norm(feats[:, None, :] - centers[None], axis=-1)
        new_labels = dist.argmin(axis=1)
        if (new_labels == labels).all():
            break
        labels = new_labels
        for c in range(k):
            if (labels == c).any():
                centers[c] = feats[labels == c].mean(axis=0)
    return labels, centers


def cluster_features(cluster_poses: list):
    """簇均值几何特征：抬手程度、腕距比、腿长（归一化躯干单位）。"""
    mean = np.mean(cluster_poses, axis=0)
    shoulder_y = (mean[10][1] + mean[11][1]) / 2
    hip_y = (mean[12][1] + mean[15][1]) / 2
    ankle_y = (mean[7][1] + mean[8][1]) / 2
    wrist_min_y = min(mean[16][1], mean[17][1])
    wrist_span = abs(mean[16][0] - mean[17][0])
    shoulder_span = abs(mean[10][0] - mean[11][0]) + 1e-6
    return {
        "wrist_raise": shoulder_y - wrist_min_y,      # 越大抬手越高
        "wrist_span_ratio": wrist_span / shoulder_span,
        "leg_drop": ankle_y - hip_y,                   # 越大腿越伸展

def assign_categories(clusters: list) -> list:
    """按簇特征排序分配类别，保证 4 个类别都被覆盖（数据驱动，免阈值）。"""
    feats = [cluster_features(c) for c in clusters]
    cats = [None] * len(clusters)
    remaining = set(range(len(clusters)))

    def take(key, reverse):
        best = max(remaining, key=lambda i: feats[i][key]) if reverse else \
            min(remaining, key=lambda i: feats[i][key])
        remaining.discard(best)
        return best

    # 抬手最高 → 手势；腕距最大 → 动态；腿最短 → 坐姿；其余 → 站姿
    cats[take("wrist_raise", True)] = "gesture"
    cats[take("wrist_span_ratio", True)] = "dynamic"
    cats[take("leg_drop", False)] = "sitting"
    for i in remaining:
        cats[i] = "standing"
    return cats


def pose_distance(a: np.ndarray, b: np.ndarray) -> float:
    return float(np.linalg.norm(a.reshape(-1) - b.reshape(-1)))


def diversify(candidates, max_keep):
    """贪心最远点采样，保证姿势库多样性。"""
    if len(candidates) <= max_keep:
        return candidates
    chosen = [candidates[0]]
    rest = candidates[1:]
    while rest and len(chosen) < max_keep:
        d = [min(pose_distance(c, k) for k in chosen) for c in rest]
        idx = int(np.argmax(d))
        chosen.append(rest.pop(idx))
    return chosen

import torch
import torch.nn as nn
from transformers import CLIPModel, CLIPProcessor


class SceneConditionedPoseVAE(nn.Module):

    def __init__(self, pose_dim=75, latent_dim=64):
        super().__init__()

        # ========= Scene Encoder (CLIP) =========
        # 提取特征信息
        self.clip = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
        self.clip_processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

        for p in self.clip.parameters():
            p.requires_grad = False

        scene_dim = 512

        # ========= Encoder =========
        # 将姿势与场景结合
        self.encoder = nn.Sequential(
            nn.Linear(pose_dim + scene_dim, 512),

----- 第 31 页 -----
            nn.ReLU(),
            nn.Linear(512, 256),
            nn.ReLU()
        )

        self.mu = nn.Linear(256, latent_dim)
        self.logvar = nn.Linear(256, latent_dim)

        # ========= Decoder =========
        # 还原姿势
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim + scene_dim, 512),
            nn.ReLU(),
            nn.Linear(512, 512),
            nn.ReLU(),
            nn.Linear(512, pose_dim)
        )

    # # 对数字归一化
    # def encode_scene(self, rgb):
    #     with torch.no_grad():
    #         inputs = self.clip_processor(images=rgb, return_tensors="pt").to(rgb.device)
    #         feat = self.clip.get_image_features(**inputs)
    #         feat = feat / feat.norm(dim=-1, keepdim=True)
    #     return feat

    def encode_scene(self, rgb):
        with torch.no_grad():
            inputs = self.clip_processor(images=rgb, return_tensors="pt").to(self.clip.device)
            feat = self.clip.get_image_features(**inputs)
            # transformers v5 起返回 BaseModelOutputWithPooling，取投影后特征
            if not torch.is_tensor(feat):
                feat = feat.pooler_output
            feat = feat / feat.norm(dim=-1, keepdim=True)
        return feat

    # 随机采样（核心要点，变化范围）
    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def forward_with_scene_features(self, pose_gt, scene_feat):
        """与训练时一致：拼接 (pose, scene) 后编码，再与 scene 解码。"""
        x = torch.cat([pose_gt, scene_feat], dim=-1)
        h = self.encoder(x)
        mu = self.mu(h)
        logvar = self.logvar(h)
        z = self.reparameterize(mu, logvar)
        dec_input = torch.cat([z, scene_feat], dim=-1)

----- 第 32 页 -----
        pose_pred = self.decoder(dec_input)
        return pose_pred, mu, logvar

    def generate_with_scene_features(self, scene_feat, num_samples=5):
        """仅依赖场景特征，从先验采样 z 生成姿态 (B, num_samples, pose_dim)。"""
        B = scene_feat.shape[0]
        z = torch.randn(B * num_samples, self.mu.out_features, device=scene_feat.device, dtype=scene_feat.dtype)
        sf = scene_feat.unsqueeze(1).expand(-1, num_samples, -1).reshape(B * num_samples, -1)
        dec_input = torch.cat([z, sf], dim=-1)
        poses = self.decoder(dec_input)
        return poses.reshape(B, num_samples, -1)

    def forward(self, rgb, pose_gt):
        scene_feat = self.encode_scene(rgb)
        return self.forward_with_scene_features(pose_gt, scene_feat)

    def generate(self, rgb, num_samples=5):
        scene_feat = self.encode_scene(rgb)
        return self.generate_with_scene_features(scene_feat, num_samples=num_samples)

def _ensure_pose_landmarker_model():
    override = os.environ.get("POSE_LANDMARKER_MODEL")
    if override and os.path.isfile(override):
        return override
    root = os.path.dirname(os.path.abspath(__file__))
    cache = os.path.join(root, ".cache", "mediapipe")
    os.makedirs(cache, exist_ok=True)
    path = os.path.join(cache, "pose_landmarker_lite.task")
    min_bytes = 5_000_000
    if not os.path.isfile(path) or os.path.getsize(path) < min_bytes:
        print(f"正在下载姿态模型到: {path}", flush=True)
        for attempt in range(3):
            try:
                urllib.request.urlretrieve(_POSE_LITE_URL, path)
                if os.path.isfile(path) and os.path.getsize(path) >= min_bytes:
                    break
            except (urllib.error.ContentTooShortError, OSError):
                if os.path.isfile(path):
                    try:
                        os.remove(path)
                    except OSError:
                        pass
                if attempt == 2:
                    raise
    return path


def _landmarks_to_simple_list(landmarks):
    class _LM:
        __slots__ = ("x", "y", "z")

        def __init__(self, x, y, z):
            self.x = float(x)
            self.y = float(y)
            self.z = float(z) if z is not None else 0.0

    return [_LM(lm.x, lm.y, getattr(lm, "z", None)) for lm in landmarks]

def extract_pose_vector_from_image(image_path, static_image_mode=True):
    import mediapipe as mp

    if hasattr(mp, "tasks") and hasattr(mp.tasks, "vision"):
        from mediapipe.tasks.python.vision.core.image import Image as MPImage
        from mediapipe.tasks.python.vision.core.image import ImageFormat

        img_bgr = _imread_unicode(image_path)
        if img_bgr is None:
            raise FileNotFoundError(f"无法读取图像: {image_path}")
        image_rgb = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)
        model_path = _ensure_pose_landmarker_model()
        landmarker = mp.tasks.vision.PoseLandmarker.create_from_model_path(model_path)
        try:
            mp_img = MPImage(ImageFormat.SRGB, image_rgb)
            result = landmarker.detect(mp_img)
        finally:
            landmarker.close()
        if not result.pose_landmarks:
            return None, None
        simple = _landmarks_to_simple_list(result.pose_landmarks[0])
        vec, joints25 = mediapipe_to_75d_vector(simple)
        return np.asarray(vec, dtype=np.float32), joints25

    if hasattr(mp, "solutions"):
        img_bgr = _imread_unicode(image_path)
        if img_bgr is None:
            raise FileNotFoundError(f"无法读取图像: {image_path}")
        image_rgb = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)
        mp_pose = mp.solutions.pose
        with mp_pose.Pose(
            static_image_mode=static_image_mode,
            model_complexity=2,
            min_detection_confidence=0.5,
        ) as pose:
            results = pose.process(image_rgb)
        if not results.pose_landmarks:
            return None, None
        vec, joints25 = mediapipe_to_75d_vector(results.pose_landmarks.landmark)
        return np.asarray(vec, dtype=np.float32), joints25

    raise RuntimeError("当前 mediapipe 安装既不包含 tasks 也不包含 solutions，无法提取姿态。")


def load_model(checkpoint_path, device, pose_dim=75, latent_dim=64):

----- 第 45 页 -----
    ckpt = torch.load(checkpoint_path, map_location=device, weights_only=False)
    args = ckpt.get("args")
    if args is not None:
        pose_dim = getattr(args, "pose_dim", pose_dim)
        latent_dim = getattr(args, "latent_dim", latent_dim)
    model = SceneConditionedPoseVAE(pose_dim=pose_dim, latent_dim=latent_dim).to(device)
    model.load_state_dict(ckpt["model_state_dict"], strict=True)
    model.eval()
    return model, pose_dim, latent_dim
