/**
 * CameraX 相机控制器：预览 + 帧分析双通道。
 * 帧分析流既驱动 MediaPipe 姿态推理，也为零快门延迟拍照提供最新帧。
 */
class CameraController(
    private val context: Context,
    private val lifecycleOwner: LifecycleOwner,
    private val previewView: PreviewView,
    private val onFrame: (androidx.camera.core.ImageProxy) -> Unit,
) {
    var lensFacing: Int = CameraSelector.LENS_FACING_BACK
        private set

    private var cameraProvider: ProcessCameraProvider? = null
    private val analysisExecutor: ExecutorService = Executors.newSingleThreadExecutor()

    fun start() {
        val future = ProcessCameraProvider.getInstance(context)
        future.addListener({
            cameraProvider = future.get()
            bind()
        }, ContextCompat.getMainExecutor(context))
    }

----- 第 10 页 -----

    fun switchCamera() {
        lensFacing = if (lensFacing == CameraSelector.LENS_FACING_BACK) {
            CameraSelector.LENS_FACING_FRONT
        } else {
            CameraSelector.LENS_FACING_BACK
        }
        bind()
    }

    fun isFront() = lensFacing == CameraSelector.LENS_FACING_FRONT

    private fun bind() {
        val provider = cameraProvider ?: return

        val preview = Preview.Builder().build().also {
            it.setSurfaceProvider(previewView.surfaceProvider)
        }

        val resolutionSelector = ResolutionSelector.Builder()
            .setResolutionStrategy(
                ResolutionStrategy(
                    android.util.Size(1280, 720),
                    ResolutionStrategy.FALLBACK_RULE_CLOSEST_HIGHER_THEN_LOWER
                )
            )
            .build()

        val analysis = ImageAnalysis.Builder()
            .setResolutionSelector(resolutionSelector)
            .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
            .setOutputImageFormat(ImageAnalysis.OUTPUT_IMAGE_FORMAT_YUV_420_888)
            .build()
        analysis.setAnalyzer(analysisExecutor) { image ->
            onFrame(image)
            image.close()
        }

        provider.unbindAll()
        provider.bindToLifecycle(
            lifecycleOwner,
            CameraSelector.Builder().requireLensFacing(lensFacing).build(),
            preview,
            analysis,
        )
    }

    fun shutdown() {
        cameraProvider?.unbindAll()
        analysisExecutor.shutdown()

    }
}

package com.poseadvisor.app.camera

import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.ImageFormat
import android.graphics.Matrix
import android.graphics.Rect
import android.graphics.YuvImage
import androidx.camera.core.ImageProxy
import java.io.ByteArrayOutputStream

/**
 * 零快门延迟拍照的核心：最新预览帧的 NV21 字节环形缓冲。
 * 帧到达时仅做一次内存拷贝（毫秒级），按下快门时无需再向相机请求新帧。
 */
class FrameBuffer {

    data class Frame(
        val nv21: ByteArray,
        val width: Int,
        val height: Int,
        val rotationDegrees: Int,
        val timestampMs: Long,
    )

    @Volatile
    var latest: Frame? = null
        private set

    /** 从 ImageProxy 拷贝 YUV_420_888 数据为 NV21 并发布为最新帧 */
    fun accept(image: ImageProxy) {
        val w = image.width
        val h = image.height
        val nv21 = ByteArray(w * h * 3 / 2)
        yuv420ToNv21(image, nv21)
        latest = Frame(nv21, w, h, image.imageInfo.rotationDegrees, System.currentTimeMillis())
    }

    /** 将最新帧解码为处理过旋转的 Bitmap（可选缩放） */
    fun decodeLatest(maxDim: Int = 0): Bitmap? {
        val f = latest ?: return null
        val yuv = YuvImage(f.nv21, ImageFormat.NV21, f.width, f.height, null)
        val out = ByteArrayOutputStream()
        yuv.compressToJpeg(Rect(0, 0, f.width, f.height), 92, out)
        var bmp = BitmapFactory.decodeByteArray(out.toByteArray(), 0, out.size()) ?: return null

----- 第 12 页 -----
        if (f.rotationDegrees != 0) {
            val m = Matrix().apply { postRotate(f.rotationDegrees.toFloat()) }
            bmp = Bitmap.createBitmap(bmp, 0, 0, bmp.width, bmp.height, m, true)
        }
        if (maxDim > 0 && maxOf(bmp.width, bmp.height) > maxDim) {
            val scale = maxDim.toFloat() / maxOf(bmp.width, bmp.height)
            val m = Matrix().apply { postScale(scale, scale) }
            bmp = Bitmap.createBitmap(bmp, 0, 0, bmp.width, bmp.height, m, true)
        }
        return bmp
    }

    companion object {
        /** 兼容任意 rowStride / pixelStride 的 YUV_420_888 → NV21 转换 */
        fun yuv420ToNv21(image: ImageProxy, out: ByteArray) {
            val w = image.width
            val h = image.height
            val yPlane = image.planes[0]
            val uPlane = image.planes[1]
            val vPlane = image.planes[2]

            // Y 平面
            var offset = 0
            val yBuffer = yPlane.buffer
            val yRowStride = yPlane.rowStride
            if (yRowStride == w) {
                yBuffer.get(out, 0, w * h)
                offset = w * h
            } else {
                val row = ByteArray(w)
                for (r in 0 until h) {
                    yBuffer.position(r * yRowStride)
                    yBuffer.get(row, 0, w)
                    System.arraycopy(row, 0, out, offset, w)
                    offset += w
                }
            }

            // VU 平面（NV21 为 V 在前）
            val chromaH = h / 2
            val chromaW = w / 2
            val vBuffer = vPlane.buffer
            val uBuffer = uPlane.buffer
            val vRowStride = vPlane.rowStride
            val uRowStride = uPlane.rowStride
            val vPixelStride = vPlane.pixelStride
            val uPixelStride = uPlane.pixelStride

            for (r in 0 until chromaH) {
                for (c in 0 until chromaW) {

----- 第 13 页 -----
                    out[offset++] = vBuffer.get(r * vRowStride + c * vPixelStride)
                    out[offset++] = uBuffer.get(r * uRowStride + c * uPixelStride)
                }
            }
        }
    }
}

package com.poseadvisor.app.pose

import kotlin.math.abs
import kotlin.math.atan2
import kotlin.math.exp
import kotlin.math.sqrt

/**
 * 姿势比对引擎：
 * 将用户实时姿态与推荐引导姿态做尺度/位置无关的归一化后，
 * 按肢体段方向角误差加权计算匹配度（0-100），并给出最大偏差部位的纠正提示。
 */
object PoseComparator {

    data class CompareResult(
        val score: Int,                       // 0-100 匹配度
        val segmentOk: BooleanArray,          // 每肢体段是否达标（用于着色）
        val hint: String?,                    // 最需要纠正的部位提示
    )

    private const val OK_THRESHOLD_RAD = 0.28f   // 单段达标阈值（约16°）
    private const val SCORE_K = 2.6f             // 误差→分数的衰减系数

    /**
     * 归一化：骨盆平移到原点，按躯干长度（骨盆→颈）缩放，消除距离与身高差异；
     * 再做人体工学比例矫正——将四肢骨骼长度校准到标准比例
     * （离线生成数据肢体偏短，不矫正则引导失真、匹配度虚低）。

----- 第 18 页 -----
     */
    fun normalize(pose: FloatArray): FloatArray {
        val out = pose.copyOf()
        val px = pose.jx(PoseModel.PELVIS)
        val py = pose.jy(PoseModel.PELVIS)
        val torso = sqrt(
            (pose.jx(PoseModel.NECK) - px) * (pose.jx(PoseModel.NECK) - px) +
                (pose.jy(PoseModel.NECK) - py) * (pose.jy(PoseModel.NECK) - py)
        ).coerceAtLeast(1e-4f)
        for (i in 0 until PoseModel.NUM_JOINTS) {
            out[i * 3] = (pose[i * 3] - px) / torso
            out[i * 3 + 1] = (pose[i * 3 + 1] - py) / torso
        }
        proportionCorrect(out)
        return out
    }

    /** 人体工学骨骼长度（躯干=1）：上臂 0.63，前臂 0.53，腿(髋→踝) 1.6 */
    private val BONE_TARGETS = listOf(
        Triple(PoseModel.SHOULDER_L, PoseModel.ELBOW_L, 0.63f),
        Triple(PoseModel.SHOULDER_R, PoseModel.ELBOW_R, 0.63f),
        Triple(PoseModel.ELBOW_L, PoseModel.WRIST_L, 0.53f),
        Triple(PoseModel.ELBOW_R, PoseModel.WRIST_R, 0.53f),
        Triple(PoseModel.HIP_L, PoseModel.ANKLE_L, 1.60f),
        Triple(PoseModel.HIP_R, PoseModel.ANKLE_R, 1.60f),
    )

    /** 沿骨骼当前方向将远端点重定位到标准骨长（姿态方向不变，仅修比例） */
    private fun proportionCorrect(p: FloatArray) {
        // 解剖修正：髋线与骨盆中心齐平（生成数据的髋点偏低，会压折腿根网格）
        p[PoseModel.HIP_L * 3 + 1] = p.jy(PoseModel.PELVIS)
        p[PoseModel.HIP_R * 3 + 1] = p.jy(PoseModel.PELVIS)
        for ((a, b, target) in BONE_TARGETS) {
            val dx = p.jx(b) - p.jx(a)
            val dy = p.jy(b) - p.jy(a)
            val len = sqrt(dx * dx + dy * dy)
            if (len < 1e-5f) continue
            // 限幅缩放系数，避免异常数据被过度拉伸
            val scale = (target / len).coerceIn(0.6f, 4.0f)
            p[b * 3] = p.jx(a) + dx * scale
            p[b * 3 + 1] = p.jy(a) + dy * scale
        }
        // 站姿约束：踝距不低于髋距 1.8 倍（与引导剪影基型一致，避免腿根网格压折）
        val hipSpan = abs(p.jx(PoseModel.HIP_L) - p.jx(PoseModel.HIP_R))
        val ankSpan = abs(p.jx(PoseModel.ANKLE_L) - p.jx(PoseModel.ANKLE_R))
        val minSpan = maxOf(hipSpan * 1.8f, 0.30f)
        if (ankSpan < minSpan) {
            val mid = (p.jx(PoseModel.ANKLE_L) + p.jx(PoseModel.ANKLE_R)) / 2
            val sign = if (p.jx(PoseModel.ANKLE_L) >= p.jx(PoseModel.ANKLE_R)) 1f else -1f
            p[PoseModel.ANKLE_L * 3] = mid + sign * minSpan / 2

----- 第 19 页 -----
            p[PoseModel.ANKLE_R * 3] = mid - sign * minSpan / 2
        }
        // 手臂约束：肘/腕不得进入躯干侧边以内（剪影基型手臂垂于体侧，穿越躯干会压折胸腰网格）
        val sides = listOf(
            Triple(PoseModel.SHOULDER_L, PoseModel.HIP_L, intArrayOf(PoseModel.ELBOW_L, PoseModel.WRIST_L)),
            Triple(PoseModel.SHOULDER_R, PoseModel.HIP_R, intArrayOf(PoseModel.ELBOW_R, PoseModel.WRIST_R)),
        )
        for ((sh, hp, armJoints) in sides) {
            val side = if (p.jx(sh) >= p.jx(PoseModel.PELVIS)) 1f else -1f
            for (jt in armJoints) {
                val denom = p.jy(hp) - p.jy(sh)
                val t = if (abs(denom) > 1e-5f) ((p.jy(jt) - p.jy(sh)) / denom).coerceIn(0f, 1f) else 0f
                val xSide = p.jx(sh) + (p.jx(hp) - p.jx(sh)) * t
                if (side * (p.jx(jt) - xSide) < 0.06f) {
                    p[jt * 3] = xSide + side * 0.06f
                }
            }
        }
    }

    /** 引导姿势展示坐标由 SkeletonOverlayView 按视图尺寸等比适配（见 fitGuideToView） */

    fun compare(userPose: FloatArray, guidePose: FloatArray): CompareResult {
        val u = normalize(userPose)
        val g = normalize(guidePose)

        val segs = PoseModel.COMPARE_SEGMENTS
        val segOk = BooleanArray(segs.size)
        var weightedSum = 0f
        var weightTotal = 0f
        var worstIdx = -1
        var worstErr = -1f

        for (i in segs.indices) {
            val s = segs[i]
            val angU = atan2(u.jy(s.b) - u.jy(s.a), u.jx(s.b) - u.jx(s.a))
            val angG = atan2(g.jy(s.b) - g.jy(s.a), g.jx(s.b) - g.jx(s.a))
            var err = abs(angU - angG)
            if (err > Math.PI.toFloat()) err = (2 * Math.PI).toFloat() - err
            segOk[i] = err < OK_THRESHOLD_RAD
            weightedSum += err * s.weight
            weightTotal += s.weight
            if (err > worstErr) { worstErr = err; worstIdx = i }
        }

        val mean = weightedSum / weightTotal
        val score = (100f * exp(-SCORE_K * mean)).toInt().coerceIn(0, 100)

        val hint = if (worstIdx >= 0 && worstErr > OK_THRESHOLD_RAD) {
            val s = segs[worstIdx]

----- 第 20 页 -----
            // 用远端关节的相对位移给出方向提示
            val dx = g.jx(s.b) - u.jx(s.b)
            val dy = g.jy(s.b) - u.jy(s.b)
            val dir = if (abs(dy) >= abs(dx)) {
                if (dy < 0) "上" else "下"
            } else {
                if (dx < 0) "左" else "右"
            }
            "请把${s.partName}向${dir}调整"
        } else null

        return CompareResult(score, segOk, hint)
    }
}

package com.poseadvisor.app.pose

import kotlin.math.abs
import kotlin.math.atan2
import kotlin.math.sqrt

/**
 * 姿势几何特征向量：用于"场景 × 姿势"智能匹配。
 * 全部在归一化空间（骨盆原点、躯干单位长）计算，与位置/身高无关。
 */
object PoseFeatures {

    data class Features(
        val armRaise: Float,      // 抬手高度（肩到最高手腕的垂直距离，越高越张扬）
        val spread: Float,        // 双臂展开度（腕距/肩距）
        val legSpread: Float,     // 双腿展开度（踝距/肩距）
        val torsoTilt: Float,     // 躯干倾斜角（弧度，0=直立，越大越动感/随性）
        val compactness: Float,   // 肢体收敛度（各关节到骨盆的平均距离，越小越收敛）
    )

----- 第 22 页 -----

    fun of(pose75: FloatArray): Features {
        val p = PoseComparator.normalize(pose75)
        val shoulderY = (p.jy(PoseModel.SHOULDER_L) + p.jy(PoseModel.SHOULDER_R)) / 2
        val wristMinY = minOf(p.jy(PoseModel.WRIST_L), p.jy(PoseModel.WRIST_R))
        val shoulderSpan = abs(p.jx(PoseModel.SHOULDER_L) - p.jx(PoseModel.SHOULDER_R)) + 1e-6f

        val spread = abs(p.jx(PoseModel.WRIST_L) - p.jx(PoseModel.WRIST_R)) / shoulderSpan
        val legSpread = abs(p.jx(PoseModel.ANKLE_L) - p.jx(PoseModel.ANKLE_R)) / shoulderSpan

        // 躯干与竖直方向的夹角
        val dx = p.jx(PoseModel.NECK) - p.jx(PoseModel.PELVIS)
        val dy = p.jy(PoseModel.NECK) - p.jy(PoseModel.PELVIS)
        val tilt = abs(atan2(dx, -dy))

        var acc = 0f
        var cnt = 0
        for (i in intArrayOf(5, 16, 17, 7, 8, 13, 14)) {
            val d = sqrt(p.jx(i) * p.jx(i) + p.jy(i) * p.jy(i))
            acc += d; cnt++
        }

        return Features(
            armRaise = shoulderY - wristMinY,
            spread = spread,
            legSpread = legSpread,
            torsoTilt = tilt,
            compactness = acc / cnt,
        )
    }
}

package com.poseadvisor.app.pose

import com.poseadvisor.app.scene.SceneClassifier

/**
 * 智能姿势推荐引擎：
 * 将场景多维特征与每个姿势的几何特征做亲和度打分，自动排序出
 * 与当前环境气质最契合的姿势候选队列，无需人工挑选。
 *
 * 匹配直觉：
 *  - 天空/开阔 → 双臂展开、腿部分开的舒展姿势
 *  - 绿地/自然 → 站立舒展、躯干略有动势的姿势
 *  - 弱光/室内 → 收敛、低重心姿势
 *  - 暖色氛围 → 适度抬手的柔和姿势
 */
object PoseRecommender {

----- 第 23 页 -----

    data class Ranked(
        val entry: PoseEntry,
        val features: PoseFeatures.Features,
        val affinity: Float,
    )

    fun rank(library: PoseLibrary, scene: SceneClassifier.SceneProfile): List<Ranked> {
        return library.poses.map { entry ->
            val f = PoseFeatures.of(entry.joints)
            Ranked(entry, f, affinity(f, scene))
        }.sortedByDescending { it.affinity }
    }

    private fun affinity(f: PoseFeatures.Features, s: SceneClassifier.SceneProfile): Float {
        var score = 0f
        // 开阔天空 ↔ 展开度
        score += s.skyRatio * (0.9f * f.spread + 0.6f * f.legSpread)
        // 自然绿意 ↔ 站立动势（腿部分开 + 适度躯干倾斜）
        score += s.greenRatio * (0.7f * f.legSpread + 0.8f * f.torsoTilt)
        // 弱光 ↔ 收敛低重心（收敛度越小越好）
        score += (1f - s.brightness).coerceIn(0f, 1f) * (1.2f - f.compactness).coerceIn(-0.5f, 1.2f)
        // 暖色氛围 ↔ 适度抬手（过度张扬反而减分）
        score += s.warmRatio * (if (f.armRaise in 0.15f..0.9f) 0.6f else 0.15f)
        // 画面活力 ↔ 整体展开
        score += s.colorfulness * 0.4f * f.spread
        // 弱光环境惩罚大倾斜（不稳）
        if (s.brightness < 0.33f) score -= 0.5f * f.torsoTilt
        return score
    }
}

/**
 * 端上姿势合成器：在库中最契合的两个姿势之间插值，并施加小扰动，
 * 再经"自然性投影"（肢体长度锁定 + 关节角度生物力学限幅），
 * 创造出姿势库中不存在、但解剖学上自然可信的新姿势。
 *
 * 自然性优先：任何不满足约束的结果自动回退为纯插值姿势。
 */

object PoseSynthesizer {

    /** 肢体段（两端关节索引 + 中文名）——长度需要锁定 */
    private val BONES = listOf(
        PoseModel.PELVIS to PoseModel.NECK,          // 躯干
        PoseModel.NECK to PoseModel.HEAD,            // 颈头
        PoseModel.SHOULDER_L to PoseModel.ELBOW_L,   // 左上臂
        PoseModel.ELBOW_L to PoseModel.WRIST_L,      // 左前臂
        PoseModel.SHOULDER_R to PoseModel.ELBOW_R,   // 右上臂
        PoseModel.ELBOW_R to PoseModel.WRIST_R,      // 右前臂
        PoseModel.HIP_L to PoseModel.ANKLE_L,        // 左腿
        PoseModel.HIP_R to PoseModel.ANKLE_R,        // 右腿
        PoseModel.SHOULDER_L to PoseModel.SHOULDER_R, // 肩宽
        PoseModel.HIP_L to PoseModel.HIP_R,           // 髋宽
    )

    /** 关节角度限幅（弧度）：[顶点, 近端, 远端, 最小角, 最大角] */
    private data class AngleLimit(val apex: Int, val from: Int, val to: Int, val minDeg: Float, val maxDeg: Float)

    private val ANGLE_LIMITS = listOf(
        AngleLimit(PoseModel.ELBOW_L, PoseModel.SHOULDER_L, PoseModel.WRIST_L, 12f, 178f),
        AngleLimit(PoseModel.ELBOW_R, PoseModel.SHOULDER_R, PoseModel.WRIST_R, 12f, 178f),
        AngleLimit(PoseModel.SHOULDER_L, PoseModel.NECK, PoseModel.ELBOW_L, 8f, 170f),
        AngleLimit(PoseModel.SHOULDER_R, PoseModel.NECK, PoseModel.ELBOW_R, 8f, 170f),
        AngleLimit(PoseModel.HIP_L, PoseModel.PELVIS, PoseModel.ANKLE_L, 5f, 172f),
        AngleLimit(PoseModel.HIP_R, PoseModel.PELVIS, PoseModel.ANKLE_R, 5f, 172f),
    )

    /**
     * 合成一个新姿势。
     * @param a, b 两个种子姿势（75维，归一化或原始均可，内部统一归一化）
     * @param lambda 插值系数（0~1，默认随机 0.3~0.7）
     * @param noise 扰动强度（躯干单位，默认 0.05）
     */
    fun synthesize(
        a: FloatArray,
        b: FloatArray,
        lambda: Float = 0.3f + Random.nextFloat() * 0.4f,
        noise: Float = 0.05f,
    ): FloatArray {
        val na = PoseComparator.normalize(a)
        val nb = PoseComparator.normalize(b)

        // 1. 插值
        val interp = FloatArray(PoseModel.POSE_DIM) { i -> na[i] * lambda + nb[i] * (1f - lambda) }

        // 2. 小扰动（只动四肢末端与肘/腕/踝，躯干与头保持稳定 = 自然感来源之一）
        val perturbed = interp.copyOf()
        val movable = intArrayOf(13, 14, 16, 17, 18, 19, 7, 8)
        for (jt in movable) {

----- 第 25 页 -----
            perturbed[jt * 3] += (Random.nextFloat() - 0.5f) * 2f * noise
            perturbed[jt * 3 + 1] += (Random.nextFloat() - 0.5f) * 2f * noise
        }

        // 3. 自然性投影：锁定肢体长度 + 关节角度限幅（迭代 2 轮收敛）
        val natural = perturbed.copyOf()
        repeat(2) {
            projectBoneLengths(natural, reference = interp)
            projectAngleLimits(natural)
        }

        // 4. 验收：段长误差与角度均达标才返回合成结果，否则回退纯插值
        return if (isNatural(natural, interp)) natural else interp
    }

    /** 将每段骨骼的远端点沿当前方向重定位到参考段长 */
    private fun projectBoneLengths(pose: FloatArray, reference: FloatArray) {
        for ((ai, bi) in BONES) {
            val ax = pose.jx(ai); val ay = pose.jy(ai)
            var dx = pose.jx(bi) - ax; var dy = pose.jy(bi) - ay
            val len = sqrt(dx * dx + dy * dy)
            if (len < 1e-6f) continue
            val refLen = sqrt(
                (reference.jx(bi) - reference.jx(ai)).let { it * it } +
                    (reference.jy(bi) - reference.jy(ai)).let { it * it }
            )
            dx /= len; dy /= len
            pose[bi * 3] = ax + dx * refLen
            pose[bi * 3 + 1] = ay + dy * refLen
        }
    }

    /** 关节角度限幅：超过范围时把远端点旋转回边界角度 */
    private fun projectAngleLimits(pose: FloatArray) {
        for (lim in ANGLE_LIMITS) {
            val apexX = pose.jx(lim.apex); val apexY = pose.jy(lim.apex)
            val a1 = atan2(pose.jy(lim.from) - apexY, pose.jx(lim.from) - apexX)
            val a2 = atan2(pose.jy(lim.to) - apexY, pose.jx(lim.to) - apexX)
            var deg = Math.toDegrees((a2 - a1).toDouble()).toFloat()
            while (deg > 180f) deg -= 360f
            while (deg < -180f) deg += 360f
            val absDeg = kotlin.math.abs(deg)
            val clamped = absDeg.coerceIn(lim.minDeg, lim.maxDeg)
            if (clamped == absDeg) continue
            // 远端点重定位：保持与 apex 距离，角度贴到边界
            val sign = if (deg >= 0) 1f else -1f
            val newA2 = a1 + Math.toRadians((clamped * sign).toDouble()).toFloat()
            val rx = pose.jx(lim.to) - apexX
            val ry = pose.jy(lim.to) - apexY
            val r = sqrt(rx * rx + ry * ry)

----- 第 26 页 -----
            pose[lim.to * 3] = apexX + cos(newA2) * r
            pose[lim.to * 3 + 1] = apexY + sin(newA2) * r
        }
    }

    /** 自然性验收：骨骼长度误差 ≤8%，且头在骨盆上方、人体不翻转 */
    private fun isNatural(pose: FloatArray, reference: FloatArray): Boolean {
        for ((ai, bi) in BONES) {
            val len = dist(pose, ai, bi)
            val refLen = dist(reference, ai, bi)
            if (refLen > 1e-4f && kotlin.math.abs(len - refLen) / refLen > 0.08f) return false
        }
        // 头必须在骨盆上方（y 向下）
        if (pose.jy(PoseModel.HEAD) > pose.jy(PoseModel.PELVIS)) return false
        return true
    }

    private fun dist(p: FloatArray, a: Int, b: Int): Float {
        val dx = p.jx(b) - p.jx(a); val dy = p.jy(b) - p.jy(a)
        return sqrt(dx * dx + dy * dy)
    }
}

/**
 * 真人剪影姿势渲染器：
 * 内置公有领域真人剪影（Wikimedia Commons, Public Domain），
 * 用 Moving Least Squares（Schaefer 2006, affine）把剪影网格
 * 变形到目标姿势，经 Canvas.drawBitmapMesh 渲染。
 * 引导人形 = 真实人体轮廓，而非几何拼接。
 */
object MeshPoseRenderer {

    private const val MESH_W = 7          // drawBitmapMesh 横向格数
    private const val MESH_H = 11         // 纵向格数
    private const val MLS_EPS = 900f      // 权重平滑（像素平方），防近锚点爆炸

    private var bitmap: Bitmap? = null
    private var basePts = FloatArray(0)   // 锚点基坐标（位图像素）N*2
    private var ruleA = IntArray(0)
    private var ruleB = IntArray(0)
    private var ruleS = FloatArray(0)

----- 第 29 页 -----
    private var ruleR = FloatArray(0)
    private var srcGrid = FloatArray(0)   // 源网格顶点（位图坐标，行优先）

    @Volatile
    private var initialized = false

    fun init(context: Context) {
        if (initialized) return
        synchronized(this) {
            if (initialized) return
            bitmap = BitmapFactory.decodeStream(context.assets.open("silhouette.png"))
            val text = context.assets.open("silhouette_anchors.json").bufferedReader().use { it.readText() }
            val arr = JSONObject(text).getJSONArray("anchors")
            val n = arr.length()
            basePts = FloatArray(n * 2)
            ruleA = IntArray(n); ruleB = IntArray(n)
            ruleS = FloatArray(n); ruleR = FloatArray(n)
            val bw = bitmap!!.width
            val bh = bitmap!!.height
            for (i in 0 until n) {
                val a = arr.getJSONObject(i)
                val base = a.getJSONArray("base")
                basePts[i * 2] = base.getDouble(0).toFloat() * bw
                basePts[i * 2 + 1] = base.getDouble(1).toFloat() * bh
                val rule = a.getJSONArray("rule")
                ruleA[i] = rule.getInt(0)
                ruleB[i] = rule.getInt(1)
                ruleS[i] = rule.getDouble(2).toFloat()
                ruleR[i] = rule.getDouble(3).toFloat()
            }
            // 源网格（行优先，(MESH_H+1) 行 (MESH_W+1) 列）
            val verts = FloatArray((MESH_W + 1) * (MESH_H + 1) * 2)
            var k = 0
            for (r in 0..MESH_H) {
                for (c in 0..MESH_W) {
                    verts[k++] = c * (bw - 1).toFloat() / MESH_W
                    verts[k++] = r * (bh - 1).toFloat() / MESH_H
                }
            }
            srcGrid = verts
            initialized = true
        }
    }

    /** 锚点目标位置（归一化姿势空间，x 已镜像对齐剪影标注方向） */
    private fun anchorTargets(pose: FloatArray): FloatArray {
        val n = ruleA.size
        val out = FloatArray(n * 2)
        for (i in 0 until n) {
            val ai = ruleA[i]; val bi = ruleB[i]

----- 第 30 页 -----
            val ax = pose.jx(ai); val ay = pose.jy(ai)
            val dx = pose.jx(bi) - ax; val dy = pose.jy(bi) - ay
            val len = sqrt(dx * dx + dy * dy)
            if (len < 1e-6f) {
                out[i * 2] = ax; out[i * 2 + 1] = ay
            } else {
                val px = -dy / len; val py = dx / len
                out[i * 2] = ax + dx * ruleS[i] + px * ruleR[i]
                out[i * 2 + 1] = ay + dy * ruleS[i] + py * ruleR[i]
            }
        }
        return out
    }

    /** Schaefer MLS-affine：把源网格顶点变形到目标锚点定义的姿势 */
    private fun mlsWarp(tgt: FloatArray): FloatArray {
        val nVerts = srcGrid.size / 2
        val n = basePts.size / 2
        val out = FloatArray(srcGrid.size)
        val w = FloatArray(n)
        for (k in 0 until nVerts) {
            val vx = srcGrid[k * 2]; val vy = srcGrid[k * 2 + 1]
            var wsum = 0f
            var psx = 0f; var psy = 0f; var qsx = 0f; var qsy = 0f
            for (i in 0 until n) {
                val dx = basePts[i * 2] - vx
                val dy = basePts[i * 2 + 1] - vy
                val wi = 1f / (dx * dx + dy * dy + MLS_EPS)
                w[i] = wi
                wsum += wi
                psx += wi * basePts[i * 2]; psy += wi * basePts[i * 2 + 1]
                qsx += wi * tgt[i * 2]; qsy += wi * tgt[i * 2 + 1]
            }
            psx /= wsum; psy /= wsum; qsx /= wsum; qsy /= wsum
            // M = (Σw Ph^T Ph)^{-1} · (Σw Ph^T Qh)
            var m2a = 0f; var m2b = 0f; var m2c = 0f
            var m1a = 0f; var m1b = 0f; var m1c = 0f; var m1d = 0f
            for (i in 0 until n) {
                val wi = w[i]
                val px = basePts[i * 2] - psx; val py = basePts[i * 2 + 1] - psy
                val qx = tgt[i * 2] - qsx; val qy = tgt[i * 2 + 1] - qsy
                m2a += wi * px * px; m2b += wi * px * py; m2c += wi * py * py
                m1a += wi * px * qx; m1b += wi * px * qy
                m1c += wi * py * qx; m1d += wi * py * qy
            }
            // 2x2 逆矩阵
            val det = m2a * m2c - m2b * m2b + 1e-8f
            val i2a = m2c / det; val i2b = -m2b / det; val i2c = m2a / det
            val ma = i2a * m1a + i2b * m1c
            val mb = i2a * m1b + i2b * m1d

----- 第 31 页 -----
            val mc = i2b * m1a + i2c * m1c
            val md = i2b * m1b + i2c * m1d
            val rx = vx - psx; val ry = vy - psy
            out[k * 2] = rx * ma + ry * mc + qsx
            out[k * 2 + 1] = rx * mb + ry * md + qsy
        }
        return out
    }

    // 变形结果缓存：引导姿势不常变，避免每帧重复 MLS 计算
    private var cachedPoseRef: FloatArray? = null
    private var cachedRectKey = 0

    /**

    fun draw(

            val pose = PoseComparator.normalize(pose75)
            for (i in 0 until PoseModel.NUM_JOINTS) pose[i * 3] = -pose[i * 3]

            val tgt = anchorTargets(pose)
            // 目标锚点（躯干单位）→ 目标矩形内像素
            var minX = Float.MAX_VALUE; var maxX = -Float.MAX_VALUE
            var minY = Float.MAX_VALUE; var maxY = -Float.MAX_VALUE
            for (i in 0 until tgt.size / 2) {
                minX = min(minX, tgt[i * 2]); maxX = max(maxX, tgt[i * 2])
                minY = min(minY, tgt[i * 2 + 1]); maxY = max(maxY, tgt[i * 2 + 1])
            }
            val spanY = (maxY - minY).coerceAtLeast(1e-3f)
            val scale = destRect.height() * 0.94f / spanY

            val dcx = destRect.centerX(); val dcy = destRect.centerY()
            for (i in 0 until tgt.size / 2) {
                tgt[i * 2] = (tgt[i * 2] - cx) * scale + dcx

----- 第 32 页 -----
                tgt[i * 2 + 1] = (tgt[i * 2 + 1] - cy) * scale + dcy
            }
            val v = mlsWarp(tgt)
            cachedPoseRef = pose75
            cachedRectKey = rectKey
            cachedVerts = v
            v
        }

        val paint = Paint(Paint.ANTI_ALIAS_FLAG or Paint.FILTER_BITMAP_FLAG).apply {
            colorFilter = PorterDuffColorFilter(tintColor, PorterDuff.Mode.SRC_IN)

    fun bitmapAspect(): Float {

def normalize_pose(joints: np.ndarray) -> np.ndarray:

def kmeans_numpy(feats: np.ndarray, k: int, iters: int = 50, seed: int = 0):
    """轻量 KMeans（k-means++ 初始化），避免引入 sklearn 依赖。"""
    rng = np.random.default_rng(seed)
    n = len(feats)
    # k-means++ 初始化
    centers = [feats[rng.integers(n)]]
    for _ in range(k - 1):
        d = np.min(np.linalg.norm(feats[:, None, :] - np.array(centers)[None], axis=-1), axis=1)
        probs = d ** 2
        probs /= probs.sum() + 1e-12
        centers.append(feats[rng.choice(n, p=probs)])
    centers = np.array(centers)
    labels = np.zeros(n, dtype=int)
    for _ in range(iters):
        dist = np.linalg.norm(feats[:, None, :] - centers[None], axis=-1)
        new_labels = dist.argmin(axis=1)
        if (new_labels == labels).all():
            break
        labels = new_labels
        for c in range(k):
            if (labels == c).any():
                centers[c] = feats[labels == c].mean(axis=0)
    return labels, centers

def assign_categories(clusters: list) -> list:
    """按簇特征排序分配类别，保证 4 个类别都被覆盖（数据驱动，免阈值）。"""
    feats = [cluster_features(c) for c in clusters]
    cats = [None] * len(clusters)
    remaining = set(range(len(clusters)))

    def take(key, reverse):
        best = max(remaining, key=lambda i: feats[i][key]) if reverse else \
            min(remaining, key=lambda i: feats[i][key])
        remaining.discard(best)
        return best

    # 抬手最高 → 手势；腕距最大 → 动态；腿最短 → 坐姿；其余 → 站姿
    cats[take("wrist_raise", True)] = "gesture"
    cats[take("wrist_span_ratio", True)] = "dynamic"
    cats[take("leg_drop", False)] = "sitting"
    for i in remaining:
        cats[i] = "standing"
    return cats


def pose_distance(a: np.ndarray, b: np.ndarray) -> float:
    return float(np.linalg.norm(a.reshape(-1) - b.reshape(-1)))


def diversify(candidates, max_keep):
    """贪心最远点采样，保证姿势库多样性。"""
    if len(candidates) <= max_keep:
        return candidates
    chosen = [candidates[0]]
    rest = candidates[1:]
    while rest and len(chosen) < max_keep:
        d = [min(pose_distance(c, k) for k in chosen) for c in rest]
        idx = int(np.argmax(d))
        chosen.append(rest.pop(idx))
    return chosen

def normalize_pose(j):
    out = j.copy()
    torso = np.linalg.norm(out[NECK, :2] - out[PELVIS, :2]) + 1e-6
    out[:, :2] = (out[:, :2] - out[PELVIS, :2]) / torso
    # 解剖修正：髋线与骨盆中心齐平（生成数据的髋点偏低，会压折腿根网格）
    out[HIP_L, 1] = out[PELVIS, 1]
    out[HIP_R, 1] = out[PELVIS, 1]
    for a, b, target in BONE_TARGETS:
        d = out[b, :2] - out[a, :2]
        l = np.linalg.norm(d)
        if l < 1e-5:
            continue
        s = np.clip(target / l, 0.6, 4.0)
        out[b, :2] = out[a, :2] + d * s
    # 站姿约束：踝距不低于髋距的 1.8 倍（与引导剪影基型一致，避免腿根网格压折）
    hip_span = abs(out[HIP_L, 0] - out[HIP_R, 0])
    ank_span = abs(out[ANKLE_L, 0] - out[ANKLE_R, 0])
    min_span = max(hip_span * 1.8, 0.30)
    if ank_span < min_span:
        mid = (out[ANKLE_L, 0] + out[ANKLE_R, 0]) / 2
        sign = 1.0 if out[ANKLE_L, 0] >= out[ANKLE_R, 0] else -1.0
        out[ANKLE_L, 0] = mid + sign * min_span / 2
        out[ANKLE_R, 0] = mid - sign * min_span / 2
    # 手臂约束：肘/腕不得进入躯干侧边以内（剪影基型手臂垂于体侧，穿越躯干会压折胸腰网格）
    for sh, hp, el, wr in ((SHOULDER_L, HIP_L, ELBOW_L, WRIST_L),
                           (SHOULDER_R, HIP_R, ELBOW_R, WRIST_R)):
        side = 1.0 if out[sh, 0] >= out[PELVIS, 0] else -1.0
        for jt in (el, wr):
            y = out[jt, 1]
            denom = out[hp, 1] - out[sh, 1]
            t = np.clip((y - out[sh, 1]) / denom, 0.0, 1.0) if abs(denom) > 1e-5 else 0.0
            x_side = out[sh, 0] + (out[hp, 0] - out[sh, 0]) * t
            if side * (out[jt, 0] - x_side) < 0.06:
                out[jt, 0] = x_side + side * 0.06
    return out


def anchor_target(rule, pose):
    a_i, b_i, s, r = rule
    A = pose[a_i, :2]
    B = pose[b_i, :2]

----- 第 48 页 -----
    d = B - A
    n = np.linalg.norm(d)
    if n < 1e-6:
        return A
    perp = np.array([-d[1], d[0]]) / n
    return A + d * s + perp * r


# ---------------------------------------------------------------- MLS affine 变形
def mls_affine(v, base_pts, tgt_pts, alpha=1.0, eps=900.0):
    """v:(M,2) 网格点 base_pts/tgt_pts:(N,2) → 变形后 (M,2)
    eps 为权重平滑项（像素平方），防止靠近锚点时权重爆炸。"""
    N = base_pts.shape[0]
    out = np.empty_like(v)
    for k in range(v.shape[0]):
        p = v[k]
        dist2 = ((base_pts - p) ** 2).sum(axis=1)
        w = 1.0 / (dist2 + eps) ** alpha
        wsum = w.sum()
        p_star = (w[:, None] * base_pts).sum(axis=0) / wsum
        q_star = (w[:, None] * tgt_pts).sum(axis=0) / wsum
        Ph = base_pts - p_star
        Qh = tgt_pts - q_star
        M1 = np.einsum("i,ij,ik->jk", w, Ph, Qh)   # Σ wi Ph^T Qh
        M2 = np.einsum("i,ij,ik->jk", w, Ph, Ph)   # Σ wi Ph^T Ph
        # Schaefer MLS affine: M = (Σw Ph^T Ph)^{-1} · (Σw Ph^T Qh)
        M = np.linalg.inv(M2 + np.eye(2) * 1e-8) @ M1
        out[k] = (p - p_star) @ M + q_star
    return out

import torch
import torch.nn as nn
from transformers import CLIPModel, CLIPProcessor


class SceneConditionedPoseVAE(nn.Module):

    def __init__(self, pose_dim=75, latent_dim=64):
        super().__init__()

        # ========= Scene Encoder (CLIP) =========
        # 提取特征信息
        self.clip = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
        self.clip_processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

        for p in self.clip.parameters():
            p.requires_grad = False

        scene_dim = 512

        # ========= Encoder =========
        # 将姿势与场景结合
        self.encoder = nn.Sequential(
            nn.Linear(pose_dim + scene_dim, 512),
            nn.ReLU(),
            nn.Linear(512, 256),
            nn.ReLU()
        )

        self.mu = nn.Linear(256, latent_dim)
        self.logvar = nn.Linear(256, latent_dim)

        # ========= Decoder =========
        # 还原姿势
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim + scene_dim, 512),
            nn.ReLU(),
            nn.Linear(512, 512),
            nn.ReLU(),
            nn.Linear(512, pose_dim)
        )

    # # 对数字归一化

----- 第 52 页 -----
    # def encode_scene(self, rgb):
    #     with torch.no_grad():
    #         inputs = self.clip_processor(images=rgb, return_tensors="pt").to(rgb.device)
    #         feat = self.clip.get_image_features(**inputs)
    #         feat = feat / feat.norm(dim=-1, keepdim=True)
    #     return feat

    def encode_scene(self, rgb):
        with torch.no_grad():
            inputs = self.clip_processor(images=rgb, return_tensors="pt").to(self.clip.device)
            feat = self.clip.get_image_features(**inputs)
            # transformers v5 起返回 BaseModelOutputWithPooling，取投影后特征
            if not torch.is_tensor(feat):
                feat = feat.pooler_output
            feat = feat / feat.norm(dim=-1, keepdim=True)
        return feat

    # 随机采样（核心要点，变化范围）
    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def forward_with_scene_features(self, pose_gt, scene_feat):
        """与训练时一致：拼接 (pose, scene) 后编码，再与 scene 解码。"""
        x = torch.cat([pose_gt, scene_feat], dim=-1)
        h = self.encoder(x)
        mu = self.mu(h)
        logvar = self.logvar(h)
        z = self.reparameterize(mu, logvar)
        dec_input = torch.cat([z, scene_feat], dim=-1)
        pose_pred = self.decoder(dec_input)
        return pose_pred, mu, logvar

    def generate_with_scene_features(self, scene_feat, num_samples=5):
        """仅依赖场景特征，从先验采样 z 生成姿态 (B, num_samples, pose_dim)。"""
        B = scene_feat.shape[0]
        z = torch.randn(B * num_samples, self.mu.out_features, device=scene_feat.device, dtype=scene_feat.dtype)
        sf = scene_feat.unsqueeze(1).expand(-1, num_samples, -1).reshape(B * num_samples, -1)
        dec_input = torch.cat([z, sf], dim=-1)
        poses = self.decoder(dec_input)
        return poses.reshape(B, num_samples, -1)

    def forward(self, rgb, pose_gt):
        scene_feat = self.encode_scene(rgb)
        return self.forward_with_scene_features(pose_gt, scene_feat)

    def generate(self, rgb, num_samples=5):
        scene_feat = self.encode_scene(rgb)
        return self.generate_with_scene_features(scene_feat, num_samples=num_samples)

POSE_DIM = NUM_JOINTS * 3


def vector_to_joints(pose_vector):
    """(75,) 或 (B,75) -> (25,3) 或 (B,25,3) numpy"""
    v = np.asarray(pose_vector, dtype=np.float32)
    if v.ndim == 1:
        assert v.shape[0] == POSE_DIM, f"期望 {POSE_DIM} 维，得到 {v.shape[0]}"
        return v.reshape(NUM_JOINTS, 3)
    assert v.ndim == 2 and v.shape[1] == POSE_DIM
    return v.reshape(v.shape[0], NUM_JOINTS, 3)


def joints_to_vector(joints):
    """(25,3) 或 (B,25,3) -> (75,) 或 (B,75)"""
    j = np.asarray(joints, dtype=np.float32)
    if j.ndim == 2:
        assert j.shape == (NUM_JOINTS, 3)
        return j.reshape(-1)
    assert j.ndim == 3 and j.shape[1:] == (NUM_JOINTS, 3)
    return j.reshape(j.shape[0], -1)


def _draw_skeleton_3d(ax, joints, line_color="C0", point_color="C0", linewidth=2, alpha=0.85):
    j = np.asarray(joints, dtype=np.float64)
    for a, b in SKELETON_CONNECTIONS:
        if a < len(j) and b < len(j):
            seg = np.stack([j[a], j[b]], axis=0)
            ax.plot(seg[:, 0], seg[:, 1], seg[:, 2], color=line_color, linewidth=linewidth, alpha=alpha)
    ax.scatter(j[:, 0], j[:, 1], j[:, 2], c=point_color, s=28, depthshade=True, alpha=0.9)

def _landmarks_to_simple_list(landmarks):
    class _LM:
        __slots__ = ("x", "y", "z")

        def __init__(self, x, y, z):
            self.x = float(x)
            self.y = float(y)
            self.z = float(z) if z is not None else 0.0

    return [_LM(lm.x, lm.y, getattr(lm, "z", None)) for lm in landmarks]


def _imread_unicode(path):
    data = np.fromfile(path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    return img


def image_path_to_pil_rgb(path):
    img_bgr = _imread_unicode(path)
    if img_bgr is None:
        raise FileNotFoundError(f"无法读取图像: {path}")
    rgb = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)
    return Image.fromarray(rgb)

def load_model(checkpoint_path, device, pose_dim=75, latent_dim=64):
    ckpt = torch.load(checkpoint_path, map_location=device, weights_only=False)
    args = ckpt.get("args")
    if args is not None:
        pose_dim = getattr(args, "pose_dim", pose_dim)
        latent_dim = getattr(args, "latent_dim", latent_dim)
    model = SceneConditionedPoseVAE(pose_dim=pose_dim, latent_dim=latent_dim).to(device)
    model.load_state_dict(ckpt["model_state_dict"], strict=True)
    model.eval()
    return model, pose_dim, latent_dim
